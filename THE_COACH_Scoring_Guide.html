1. ໂຄ້ດ Google Apps Script (Code.gs)

ກອບໂຄ້ດນີ້ໄປວາງໃນ Extensions → Apps Script ໃນ Google Sheets:

/** * ============================================ * ໂຄງການ SIMPLE - ແບບຟອມເກັບຄະແນນ THE COACH * ============================================ */ function createCoachScoringSheet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheetName = "THE COACH - ເກັບຄະແນນ"; // ລຶບ Sheet ເກົ່າຖ້າມີຊື່ຊ້ຳ var existingSheet = ss.getSheetByName(sheetName); if (existingSheet) { ss.deleteSheet(existingSheet); } var sheet = ss.insertSheet(sheetName); // ===== ຕັ້ງຄ່າຂະໜາດຖັນ ===== sheet.setColumnWidth(1, 320); sheet.setColumnWidth(2, 110); sheet.setColumnWidth(3, 130); sheet.setColumnWidth(4, 130); // ===== 1. ສ່ວນຫົວຂໍ້ (Rows 1-3) ===== sheet.getRange("A1:D1").merge() .setValue("🎯 ແບບຟອມເກັບຄະແນນ - ໂຄງການ SIMPLE (THE COACH)") .setFontSize(16) .setFontWeight("bold") .setHorizontalAlignment("center") .setBackground("#1a237e") .setFontColor("#ffffff"); sheet.setRowHeight(1, 40); // Row 2: ຂໍ້ມູນທົ່ວໄປ sheet.getRange("A2").setValue("👥 ຊື່ກຸ່ມ:").setFontWeight("bold"); sheet.getRange("B2").setValue("").setBackground("#e3f2fd"); sheet.getRange("C2").setValue("🎮 ຮອບການຫຼິ້ນ:").setFontWeight("bold"); sheet.getRange("D2").setValue("").setBackground("#e3f2fd"); // Row 3: ສະມາຊິກ sheet.getRange("A3").setValue("📝 ລາຍຊື່ສະມາຊິກ:").setFontWeight("bold"); sheet.getRange("B3:D3").merge() .setValue("").setBackground("#e3f2fd") .setWrap(true); sheet.setRowHeight(3, 30); // ===== 2. ສ່ວນຄະແນນດ້ານບວກ (HERO SCORE) ===== sheet.getRange("A5:D5").merge() .setValue("🦸 ຄະແນນດ້ານບວກ (HERO SCORE)") .setFontSize(13).setFontWeight("bold") .setHorizontalAlignment("center") .setBackground("#2e7d32").setFontColor("#ffffff"); sheet.setRowHeight(5, 32); // Headers var headers = ["ກິດຈະກໍາ / ພາລະກິດ", "ຄະແນນ", "ຈຳນວນຄັ້ງ", "ຄະແນນລວມ"]; sheet.getRange("A6:D6").setValues([headers]) .setFontWeight("bold").setBackground("#c8e6c9") .setFontColor("#1b5e20"); // Row 7: ເກັບຕົວຢ່າງ (+100) sheet.getRange("A7").setValue("🔬 ເກັບຕົວຢ່າງວິທະຍາສາດຖືກຕ້ອງ"); sheet.getRange("B7").setValue(100).setHorizontalAlignment("center"); sheet.getRange("C7").setValue("").setBackground("#fff9c4").setHorizontalAlignment("center"); sheet.getRange("D7").setFormula("=B7*C7").setFontWeight("bold"); // Row 8: ຊ່ວຍຄົນເຈັບ (+50) sheet.getRange("A8").setValue("🚑 ຊ່ວຍເຫຼືອຄົນເຈັບທັນເວລາ"); sheet.getRange("B8").setValue(50).setHorizontalAlignment("center"); sheet.getRange("C8").setValue("").setBackground("#fff9c4").setHorizontalAlignment("center"); sheet.getRange("D8").setFormula("=B8*C8").setFontWeight("bold"); // Row 9: ໂບນັດ O2 (+200, 1/0) sheet.getRange("A9").setValue("💨 ໂບນັດອອກຊີເຈນເຫຼືອ >50%"); sheet.getRange("B9").setValue(200).setHorizontalAlignment("center"); sheet.getRange("C9").setValue("").setBackground("#fff9c4").setHorizontalAlignment("center"); // Data validation: ໃສ່ໄດ້ສະເພາະ 0 ຫຼື 1 var ruleO2 = SpreadsheetApp.newDataValidation() .requireNumberBetween(0, 1) .setAllowInvalid(false) .setHelpText("ກະລຸນາໃສ່ 0 ຫຼື 1 ເທົ່ານັ້ນ") .build(); sheet.getRange("C9").setDataValidation(ruleO2); sheet.getRange("D9").setFormula("=B9*C9").setFontWeight("bold"); // Row 10: ລວມຄະແນນສີຂຽວ sheet.getRange("A10:C10").merge().setValue("📗 ລວມຄະແນນດ້ານບວກທັງໝົດ") .setFontWeight("bold").setHorizontalAlignment("right") .setFontColor("#1b5e20"); sheet.getRange("D10").setFormula("=SUM(D7:D9)") .setFontWeight("bold").setFontSize(12) .setBackground("#66bb6a").setFontColor("#ffffff"); // ===== 3. ສ່ວນຄະແນນດ້ານລົບ (PENALTY) ===== sheet.getRange("A12:D12").merge() .setValue("⚠️ ຄະແນນດ້ານລົບ (PENALTY SCORE)") .setFontSize(13).setFontWeight("bold") .setHorizontalAlignment("center") .setBackground("#c62828").setFontColor("#ffffff"); sheet.setRowHeight(12, 32); sheet.getRange("A13:D13").setValues([headers]) .setFontWeight("bold").setBackground("#ffcdd2") .setFontColor("#b71c1c"); // Row 14: ເຊື້ອແຜ່ລະບາດ (-80) sheet.getRange("A14").setValue("🛡️ ປ້ອງກັນຜິດພາດ / ເຊື້ອແຜ່ລະບາດ"); sheet.getRange("B14").setValue(80).setHorizontalAlignment("center"); sheet.getRange("C14").setValue("").setBackground("#fff9c4").setHorizontalAlignment("center"); sheet.getRange("D14").setFormula("=B14*C14").setFontWeight("bold"); // Row 15: ວິນິດໄສຜິດ (-150) sheet.getRange("A15").setValue("❌ ວິນິດໄສຜິດ / ສົ່ງຂໍ້ມູນຜິດ"); sheet.getRange("B15").setValue(150).setHorizontalAlignment("center"); sheet.getRange("C15").setValue("").setBackground("#fff9c4").setHorizontalAlignment("center"); sheet.getRange("D15").setFormula("=B15*C15").setFontWeight("bold"); // Row 16: ລວມຄະແນນສີແດງ sheet.getRange("A16:C16").merge().setValue("📕 ລວມຄະແນນດ້ານລົບທັງໝົດ") .setFontWeight("bold").setHorizontalAlignment("right") .setFontColor("#b71c1c"); sheet.getRange("D16").setFormula("=SUM(D14:D15)") .setFontWeight("bold").setFontSize(12) .setBackground("#ef5350").setFontColor("#ffffff"); // ===== 4. ສະຫຼຸບຜົນ ===== sheet.getRange("A18:D18").merge() .setValue("📊 ສະຫຼຸບຜົນສຸດທ້າຍ") .setFontSize(13).setFontWeight("bold") .setHorizontalAlignment("center") .setBackground("#1565c0").setFontColor("#ffffff"); sheet.getRange("A19:C19").merge().setValue("🏆 ຄະແນນສຸດທ້າຍ") .setFontWeight("bold").setHorizontalAlignment("right") .setBackground("#e3f2fd"); sheet.getRange("D19").setFormula("=D10-D16") .setFontWeight("bold").setFontSize(14) .setBackground("#1976d2").setFontColor("#ffffff"); sheet.getRange("A20:C20").merge().setValue("📋 ລະດັບການປະເມີນ") .setFontWeight("bold").setHorizontalAlignment("right") .setBackground("#e3f2fd"); sheet.getRange("D20").setFormula('=IF(D19>=500,"🟢 ລະດັບ ທຸດຊຸມຊົນ (ດີເລີດ)",IF(D19>=250,"🟡 ລະດັບ ຜູ້ປ້ອງກັນ (ປານກາງ)","🔴 ລະດັບ ມືໃໝ່ (ຕ້ອງປັບປຸງ)"))') .setFontWeight("bold").setFontSize(12) .setBackground("#90caf9"); // ===== 5. ຂອບ + ຟ້ອນ + ປ້ອງກັນສູດ ===== var allDataRange = sheet.getRange("A1:D20"); allDataRange.setBorder(true, true, true, true, true, true, "#000000", SpreadsheetApp.BorderStyle.SOLID); allDataRange.setFontFamily("Arial"); allDataRange.setFontSize(11); allDataRange.setVerticalAlignment("middle"); // ===== 6. Conditional Formatting ===== var rangeD20 = sheet.getRange("D20"); var rules = []; var rule1 = SpreadsheetApp.newConditionalFormatRule() .whenTextContains("ທຸດຊຸມຊົນ") .setBackground("#c8e6c9").setFontColor("#1b5e20") .setRanges([rangeD20]).build(); rules.push(rule1); var rule2 = SpreadsheetApp.newConditionalFormatRule() .whenTextContains("ຜູ້ປ້ອງກັນ") .setBackground("#fff9c4").setFontColor("#f57f17") .setRanges([rangeD20]).build(); rules.push(rule2); var rule3 = SpreadsheetApp.newConditionalFormatRule() .whenTextContains("ມືໃໝ່") .setBackground("#ffcdd2").setFontColor("#b71c1c") .setRanges([rangeD20]).build(); rules.push(rule3); sheet.setConditionalFormatRules(rules); // ===== 7. ປ້ອງກັນສູດ ===== var formulaRanges = ["D7:D10", "D14:D16", "D19:D20"]; formulaRanges.forEach(function(rangeNotation) { var range = sheet.getRange(rangeNotation); var protection = range.protect().setDescription("ປ້ອງກັນສູດຄຳນວນ"); var editors = protection.getEditors(); editors.forEach(function(editor) { protection.removeEditor(editor); }); }); // ===== 8. ແຈ້ງເຕືອນ ===== SpreadsheetApp.getUi().alert( "✅ ສຳເລັດແລ້ວ!", "ແບບຟອມເກັບຄະແນນ THE COACH ຖືກສ້າງໃນ Sheet: '" + sheetName + "'", SpreadsheetApp.getUi().ButtonSet.OK ); }
💡 ຄຳແນະນຳ: ກອບໂຄ້ດທັງໝົດນີ້ໄປວາງໃນ Apps Script ແລ້ວກົດ Run (▶️) ຄັ້ງດຽວ ແບບຟອມຈະສ້າງອັດຕະໂນມັດ!

4. ເງື່ອນໄຂລະດັບການປະເມີນ

5. ຄຳແນະນຳເພີ່ມເຕີມ
  • ຟ້ອນພາສາລາວ: ຖ້າຕົວອັກສອນລາວບໍ່ສະແດງສົມບູນ, ໄປທີ່ Format → Font → ເລືອກ "Noto Sans Lao" ຫຼື "Phetsarath OT"
  • ສຳເນົາໃຫ້ນັກຮຽນ: ໃຊ້ File → Make a copy ແລ້ວແຈກຈ່າຍ Link ໃຫ້ແຕ່ລະກຸ່ມ
  • ເກັບປະຫວັດ: ເມື່ອຈົບແຕ່ລະຮອບ, ກົດ File → Version history → Name current version ເພື່ອເກັບຄະແນນແຕ່ລະຮອບ
